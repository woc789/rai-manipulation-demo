cmake_minimum_required(VERSION 3.16)
project(produce_order_bt)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(produce_order_interfaces REQUIRED)
add_executable(fulfillment_tree src/fulfillment_tree.cpp)
target_compile_features(fulfillment_tree PRIVATE cxx_std_17)
ament_target_dependencies(fulfillment_tree rclcpp rclcpp_action ament_index_cpp produce_order_interfaces)
target_link_libraries(fulfillment_tree behaviortree_cpp::behaviortree_cpp)
install(TARGETS fulfillment_tree DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY trees DESTINATION share/${PROJECT_NAME})
ament_package()
